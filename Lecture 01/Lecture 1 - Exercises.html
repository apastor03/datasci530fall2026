import numpy as np

# 1. Create an array of 1000 entries, all equal to one
ones_vec = np.ones(1000)

# 2. Append ones_vec and vec_x (from question f) into a matrix with 2 columns and 1000 rows
# (Assuming vec_x is already defined as a 1D array or column vector of size 1000)
X_mat = np.column_stack((ones_vec, vec_x))

# 3. Compute (X_mat' * X_mat)^-1
# X_mat.T is the transpose, and np.linalg.inv computes the inverse
X_transpose_X = np.dot(X_mat.T, X_mat)
X_mat_inv = np.linalg.inv(X_transpose_X)
